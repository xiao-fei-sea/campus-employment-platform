def generate_students(db: Session, count: int = 5000):
    """生成学生数据"""
    print(f"\n{'='*60}")
    print(f"📚 正在生成 {count:,} 条学生数据...")
    print(f"{'='*60}")
    
    students = []
    graduation_years = [2024, 2025, 2026]
    employment_statuses = [
        EmploymentStatus.SIGNED,
        EmploymentStatus.OFFERED,
        EmploymentStatus.SEARCHING,
        EmploymentStatus.UNEMPLOYED,
        EmploymentStatus.FURTHER_STUDY,
        EmploymentStatus.STARTUP
    ]
    
    # 确保有足够的就业状态分布
    status_weights = {
        EmploymentStatus.SIGNED: 0.40,  # 40% 已签约
        EmploymentStatus.OFFERED: 0.15,  # 15% 已offer
        EmploymentStatus.SEARCHING: 0.25,  # 25% 求职中
        EmploymentStatus.UNEMPLOYED: 0.08,  # 8% 未就业
        EmploymentStatus.FURTHER_STUDY: 0.10,  # 10% 继续深造
        EmploymentStatus.STARTUP: 0.02  # 2% 创业
    }
    
    for i in range(count):
        if (i + 1) % 500 == 0:
            print(f"  已生成 {i+1:,} 条学生数据...")
        
        # 创建用户账户
        user = User(
            username=f"student{i+1:05d}",
            email=f"student{i+1:05d}@campus.edu.cn",
            hashed_password=get_password_hash("password123"),
            role=UserRole.STUDENT,
            is_active=True,
            is_verified=True
        )
        db.add(user)
        db.flush()
        
        # 根据权重选择就业状态
        status = random.choices(
            list(status_weights.keys()),
            weights=list(status_weights.values())
        )[0]
        
        # 创建学生信息
        student = Student(
            user_id=user.id,
            student_no=f"{random.randint(2020, 2023)}{random.randint(10000000, 99999999)}",
            name=random_name(),
            gender=random.choice(["男", "女"]),
            phone=random_phone(),
            major=random_major(),
            college=random_college(),
            class_name=random_class_name(),
            graduation_year=random.choice(graduation_years),
            employment_status=status,
            is_key_assistance=random.random() < 0.08,  # 8% 需要帮扶
            assistance_note="家庭经济困难" if random.random() < 0.05 else None
        )
        db.add(student)
        students.append(student)
        
        # 每500条提交一次
        if (i + 1) % 500 == 0:
            db.commit()
    
    db.commit()
    print(f"  ✅ 学生数据生成完成！共 {len(students):,} 条")
    return students

            EmploymentStatus.SIGNED: {'pending': 0.1, 'viewed': 0.2, 'passed': 0.3, 'failed': 0.3, 'offered': 0.3},
            EmploymentStatus.OFFERED: {'pending': 0.15, 'viewed': 0.25, 'passed': 0.3, 'failed': 0.2, 'offered': 0.1},

            EmploymentStatus.STARTUP: {'pending': 0.3, 'viewed': 0.3, 'passed': 0.25, 'failed': 0.1, 'offered': 0.05}
        }
        
        weights = status_weights.get(student.employment_status, status_weights[EmploymentStatus.SEARCHING])
        status = random.choices(
            list(weights.keys()),
            weights=list(weights.values())
        )[0]
        
        applied_at = datetime.now() - timedelta(days=random.randint(1, 180))
        
        application = Application(
            student_id=student.id,
            job_id=job.id,
            resume_id=resume.id,
            status=ApplicationStatus(status),

        )
        db.add(application)
        count += 1
        
        if (i + 1) % 1000 == 0:
            db.commit()
    
    db.commit()
    print(f"  ✅ 投递记录生成完成！共 {count:,} 条")
    return count
